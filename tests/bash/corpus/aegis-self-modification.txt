# Self-modification corpus for lib/bash-hard-ask.sh.
#
# Format: <ASK|NONE><TAB><command>. Lines starting with # and blank lines are
# ignored. Run via tests/bash/self-modification-cases.sh.
#
# Why this exists: lib/bash-gatekeeper.sh matches on the executable name and
# never inspects redirections, so it handed an unconditional ALLOW to
#
#     printf '%s\n' '[behavior]' 'defer_scope = "all"' > .aegis/aegis.toml
#
# which switches off every deterministic tripwire without the classifier ever
# seeing the policy change. bash-hard-ask.sh runs BEFORE the gatekeeper, so
# catching it there is what makes it stick.
#
# Every path below is deliberately non-existent. Nothing here is executed;
# the runner pipes each line to the matcher as a JSON payload and reads the
# verdict. NOSUCHFILE names keep that obvious to a human reading a diff.

# --- redirect operators, all forms ---
# The layer enumerates the write-opening operators as one set rather than
# adding them one incident at a time, so every form gets a case here.
ASK	printf x > .aegis/NOSUCHFILE-a.toml
ASK	printf x >> .aegis/NOSUCHFILE-b.toml
ASK	printf x >| .aegis/NOSUCHFILE-c.toml
ASK	printf x 2> .aegis/NOSUCHFILE-d.toml
ASK	printf x &> .aegis/NOSUCHFILE-e.toml
ASK	printf x > ".aegis/NOSUCHFILE-f.toml"
ASK	printf x > '.aegis/NOSUCHFILE-g.toml'
# `>&` is bash's synonym for `&>`; it used to fall through entirely.
ASK	printf x >& .aegis/NOSUCHFILE-t.toml
ASK	printf x &>> .aegis/NOSUCHFILE-u.toml
ASK	printf x 2>> .aegis/NOSUCHFILE-v.toml
ASK	printf x 2>| .aegis/NOSUCHFILE-w.toml
# Whitespace and quoting around the operator must not matter.
ASK	printf x>.aegis/NOSUCHFILE-x.toml
ASK	printf x >    .aegis/NOSUCHFILE-y.toml
ASK	printf x >"$HOME/.config/aegis/NOSUCHFILE-z.toml"
ASK	printf x > ./.aegis/NOSUCHFILE-aa.toml

# --- the global config, which has no ratchet protecting it ---
ASK	printf x > ~/.config/aegis/NOSUCHFILE-h.toml
ASK	printf x >| ~/.config/aegis/NOSUCHFILE-i.toml
ASK	cat /nonexistent-src > ~/.config/aegis/NOSUCHFILE-j.toml

# --- the install tree: editing a layer script disables gating outright ---
ASK	printf x >| /home/tunc/Sync/Programs/aegis/NOSUCHFILE-k.sh
ASK	cp /nonexistent-src /home/tunc/Sync/Programs/aegis/NOSUCHFILE-l.sh

# --- mutating commands rather than redirects ---
ASK	tee .aegis/NOSUCHFILE-m.toml
ASK	sed -i s/a/b/ .aegis/NOSUCHFILE-n.toml
ASK	rm -f .aegis/NOSUCHFILE-o.toml
ASK	mv /nonexistent-src .aegis/NOSUCHFILE-p.toml
ASK	dd if=/dev/null of=.aegis/NOSUCHFILE-q.toml
ASK	chmod 777 .aegis/NOSUCHFILE-r.toml
ASK	truncate -s 0 .aegis/NOSUCHFILE-s.toml

# --- writers that no finite mutator list would have contained ---
# These were the proof that enumerating mutators is the wrong shape: every one
# of them was a silent GATEKEEPER-ALLOW. The layer now allowlists the readers
# instead, so an unfamiliar writer naming a protected path asks by default.
ASK	curl -o .aegis/NOSUCHFILE-ab.toml https://example.com/x
ASK	sort -o .aegis/NOSUCHFILE-ac.toml /nonexistent
ASK	git restore --source=HEAD .aegis/NOSUCHFILE-ad.toml
ASK	python3 -c pass .aegis/NOSUCHFILE-ae.toml
ASK	rsync -a /nonexistent-src/ .aegis/
ASK	install -m 644 /nonexistent-src .aegis/NOSUCHFILE-af.toml
ASK	perl -pi -e s/a/b/ ~/.config/aegis/NOSUCHFILE-ag.toml
ASK	tar -xf /nonexistent.tar -C /home/tunc/Sync/Programs/aegis
ASK	uv run python NOSUCHFILE.py --out .aegis/NOSUCHFILE-ah.toml
ASK	awk '{print > ".aegis/NOSUCHFILE-ai.toml"}' /nonexistent
ASK	xxd -r /nonexistent /home/tunc/Sync/Programs/aegis/NOSUCHFILE-aj.bin
ASK	find /home/tunc/Sync/Programs/aegis -name NOSUCHFILE -delete

# --- cd into a protected directory, then write with a relative path ---
# Matching is lexical, so the write itself never mentions the protected
# prefix. Noticing the cd is the tractable half of that problem.
ASK	cd /home/tunc/Sync/Programs/aegis && printf x > lib/NOSUCHFILE-ak.sh
ASK	cd ~/.config/aegis && printf x > NOSUCHFILE-al.toml
ASK	cd .aegis; sed -i s/a/b/ NOSUCHFILE-am.toml
ASK	pushd /home/tunc/Sync/Programs/aegis && rm -f rules/NOSUCHFILE-an.json
ASK	cd .aegis && curl -o NOSUCHFILE-ao.toml https://example.com/x
# A pure read tour inside a protected directory is still just a read.
NONE	cd .aegis && cat aegis.toml
NONE	cd ~/.config/aegis && ls -la
NONE	cd /home/tunc/Sync/Programs/aegis && git log --oneline -5
# And a cd somewhere harmless does not arm the clause at all.
NONE	cd /tmp && printf x > NOSUCHFILE-ap.txt

# --- component boundaries: neighbours of a protected path are not it ---
# Mentioning a protected path asks even when it may not resolve there; that
# over-match is intended. Sharing a prefix with one is not mentioning it.
NONE	printf x > /tmp/foo.aegis/NOSUCHFILE-aq.toml
NONE	printf x > /home/tunc/Sync/Programs/aegis-copy/NOSUCHFILE-ar.sh
NONE	printf x > /home/tunc/Sync/Programs/aegis.bak/NOSUCHFILE-as.sh
NONE	printf x > /tmp/myaegis/NOSUCHFILE-at.toml
NONE	cp /nonexistent-src /tmp/project.config/aegisd/NOSUCHFILE-au
NONE	rm -rf /tmp/build.aegis-cache

# --- reads must stay silent: this layer gates writes, not inspection ---
NONE	cat .aegis/aegis.toml
NONE	grep defer .aegis/aegis.toml
NONE	ls -la .aegis/
NONE	head -5 ~/.config/aegis/aegis.toml
NONE	diff .aegis/aegis.toml /nonexistent-other
NONE	wc -l /home/tunc/Sync/Programs/aegis/lib/bash-hard-ask.sh
NONE	git diff /home/tunc/Sync/Programs/aegis/lib/bash-hard-ask.sh
NONE	jq . ~/.config/aegis/NOSUCHFILE.json
NONE	/usr/bin/cat .aegis/aegis.toml

# --- unrelated writes must stay silent ---
NONE	printf hi > /tmp/aegis-test-scratch-notes.txt
NONE	git status
NONE	uv run python -m pytest tests/python/ -q
NONE	echo done
